package models;

import interfaces.Operatable;
import java.io.Serializable;

public abstract class FactoryEquipment implements Operatable, Serializable {
    private static final long serialVersionUID = 1L;

    protected String id;
    protected String name;
    protected boolean status;
    protected double temperature;

    public FactoryEquipment(String id, String name, double temperature) {
        this.id = id;
        this.name = name;
        this.temperature = temperature;
        this.status = false;
    }

    public abstract String getDetails();

    @Override
    public void startWork() { this.status = true; }

    @Override
    public void stopWork() { this.status = false; }

    @Override
    public boolean isWorking() { return status; }

    public String getId() { return id; }
    public void setId(String id) { this.id = id; }

    public String getName() { return name; }
    public void setName(String name) { this.name = name; }

    public boolean isStatus() { return status; }
    public void setStatus(boolean status) { this.status = status; }

    public double getTemperature() { return temperature; }
    public void setTemperature(double temperature) { this.temperature = temperature; }
}

src/logic/SafetyProtocol.java
package logic;

import models.FactoryEquipment;
import java.util.function.Consumer;
import java.util.function.Predicate;

public class SafetyProtocol<T extends FactoryEquipment> {
    private String protocolName;
    private Predicate<T> condition;
    private Consumer<T> action;

    public SafetyProtocol(String protocolName, Predicate<T> condition, Consumer<T> action) {
        this.protocolName = protocolName;
        this.condition = condition;
        this.action = action;
    }

    public void apply(T equipment) {
        if (condition.test(equipment)) {
            action.accept(equipment);
        }
    }

    public String getProtocolName() { return protocolName; }
}

src/logic/FactoryManager.java
package logic;

import models.FactoryEquipment;

import java.io.*;
import java.util.*;
import java.util.stream.Stream;

public class FactoryManager {
    private Map<String, List<FactoryEquipment>> workshops;
    private transient List<SafetyProtocol<? extends FactoryEquipment>> protocols;

    public FactoryManager() {
        this.workshops = new HashMap<>();
        this.protocols = new ArrayList<>();
    }

    public void addEquipment(String workshopName, FactoryEquipment equipment) {
        workshops.computeIfAbsent(workshopName, k -> new ArrayList<>()).add(equipment);
    }

    public void addProtocol(SafetyProtocol<?> protocol) {
        protocols.add(protocol);
    }

    public FactoryEquipment getEquipmentById(String id) {
        return workshops.values().stream()
                .flatMap(List::stream)
                .filter(e -> e.getId().equals(id))
                .findFirst()
                .orElse(null);
    }

    @SuppressWarnings("unchecked")
    public void applyAllProtocols() {
        for (List<FactoryEquipment> list : workshops.values()) {
            for (FactoryEquipment equipment : list) {
                for (SafetyProtocol<?> protocol : protocols) {
                    try {
                        ((SafetyProtocol<FactoryEquipment>) protocol).apply(equipment);
                    } catch (ClassCastException ignored) {
                        // Протокол не подходит по типу — пропускаем
                    }
                }
            }
        }
        System.out.println("Все протоколы применены.");
    }

    public Stream<FactoryEquipment> getAnalyticsStream() {
        return workshops.values().stream().flatMap(List::stream);
    }

    public Map<String, List<FactoryEquipment>> getWorkshops() {
        return workshops;
    }

    // ================= БОНУС: Сериализация =================

    public void saveState(String filename) {
        try (ObjectOutputStream oos = new ObjectOutputStream(new FileOutputStream(filename))) {
            oos.writeObject(workshops);
            System.out.println("Состояние завода сохранено в " + filename);
        } catch (IOException e) {
            System.out.println("Ошибка сохранения: " + e.getMessage());
        }
    }

    @SuppressWarnings("unchecked")
    public void loadState(String filename) {
        File file = new File(filename);
        if (!file.exists()) {
            System.out.println("Файл состояния не найден. Создан пустой завод.");
            return;
        }
        try (ObjectInputStream ois = new ObjectInputStream(new FileInputStream(file))) {
            workshops = (Map<String, List<FactoryEquipment>>) ois.readObject();
            if (protocols == null) {
                protocols = new ArrayList<>();
            }
            System.out.println("Состояние завода загружено из " + filename);
        } catch (IOException | ClassNotFoundException e) {
            System.out.println("Ошибка загрузки: " + e.getMessage());
        }
    }
}

package main;

import logic.FactoryManager;
import logic.SafetyProtocol;
import models.*;

import java.util.*;
import java.util.stream.Collectors;

public class Main {

    private static int variant = 1;
    private static final String STATE_FILE = "factory_state.dat";

    public static void main(String[] args) {
        FactoryManager manager = new FactoryManager();
        manager.loadState(STATE_FILE);

        Scanner scanner = new Scanner(System.in);

        // ---- Выбор варианта ----
        System.out.println("=== Система управления «Умным заводом» ===");
        System.out.println("Выберите вариант (1–8):");
        for (int i = 1; i <= 8; i++) {
            System.out.println(i + ". " + getVariantDescription(i));
        }
        System.out.print("Ваш вариант: ");
        try {
            variant = Integer.parseInt(scanner.nextLine().trim());
            if (variant < 1 || variant > 8) variant = 1;
        } catch (NumberFormatException e) {
            variant = 1;
        }
        System.out.println("Выбран вариант " + variant + ": " + getVariantDescription(variant));

        // ---- Главный цикл ----
        boolean running = true;
        while (running) {
            printMenu();
            String choice = scanner.nextLine().trim();

            switch (choice) {
                case "1": addEquipment(scanner, manager); break;
                case "2": toggleEquipment(scanner, manager); break;
                case "3": addPhantomHeatProtocol(manager); break;
                case "4": addVariantProtocol(manager); break;
                case "5": manager.applyAllProtocols(); break;
                case "6": runVariantAnalytics(manager); break;
                case "7":
                    manager.saveState(STATE_FILE);
                    System.out.println("Выход из программы.");
                    running = false;
                    break;
                default:
                    System.out.println("Неверный пункт. Попробуйте снова.");
            }
        }
        scanner.close();
    }

        System.out.println("\n--- Меню ---");

        System.out.println("4. Добавить протокол варианта " + variant);
        System.out.println("5. Запустить проверку всех протоколов");
        System.out.println("6. Аналитика варианта " + variant);
        System.out.println("7. Сохранить и выйти");
        System.out.print("Выберите пункт: ");
    }

    // ================= ПУНКТ 1 =================
    private static void addEquipment(Scanner scanner, FactoryManager manager) {
        System.out.print("Тип (1-RoboticArm, 2-ConveyorBelt, 3-HydraulicPress): ");
        String type = scanner.nextLine().trim();

        System.out.print("ID: ");
        String id = scanner.nextLine().trim();
        System.out.print("Имя: ");
        String name = scanner.nextLine().trim();
        System.out.print("Цех: ");
        String workshop = scanner.nextLine().trim();

        double temperature = readDouble(scanner, "Температура (°C): ");

        FactoryEquipment equipment = null;
        switch (type) {
            case "1":
                double payload = readDouble(scanner, "Нагрузка (кг): ");
                equipment = new RoboticArm(id, name, temperature, payload);
                break;
            case "2":
                double speed = readDouble(scanner, "Скорость (м/с): ");
                equipment = new ConveyorBelt(id, name, temperature, speed);
                break;
            case "3":
                double pressure = readDouble(scanner, "Давление (МПа): ");
                equipment = new HydraulicPress(id, name, temperature, pressure);
                break;
            default:
                System.out.println("Неверный тип.");
                return;
        }

        manager.addEquipment(workshop, equipment);
        System.out.println("Оборудование добавлено в цех «" + workshop + "».");
    }

    private static double readDouble(Scanner scanner, String prompt) {
        System.out.print(prompt);
        try {
            return Double.parseDouble(scanner.nextLine().trim().replace(',', '.'));
        } catch (NumberFormatException e) {
            System.out.println("Некорректное число, установлено 0.0");
            return 0.0;
        }
    }

    // ================= ПУНКТ 2 =================
    private static void toggleEquipment(Scanner scanner, FactoryManager manager) {
        System.out.print("ID оборудования: ");
        String id = scanner.nextLine().trim();
        FactoryEquipment eq = manager.getEquipmentById(id);
        if (eq == null) {
            System.out.println("Оборудование не найдено.");
            return;
        }
        System.out.println("Текущий статус: " + (eq.isWorking() ? "Работает" : "Остановлено"));
        System.out.print("Команда (start/stop): ");
        String cmd = scanner.nextLine().trim().toLowerCase();
        if (cmd.equals("start")) {
            eq.startWork();
            System.out.println("Оборудование запущено.");
        } else if (cmd.equals("stop")) {
            eq.stopWork();
            System.out.println("Оборудование остановлено.");
        } else {
            System.out.println("Неверная команда.");
        }
    }

    private static void addPhantomHeatProtocol(FactoryManager manager) {
        SafetyProtocol<FactoryEquipment> phantom = new SafetyProtocol<>(
                "Сброс фантомного нагрева",
                eq -> !eq.isWorking() && eq.getTemperature() > 20.0,
                eq -> eq.setTemperature(20.0)
        );
        manager.addProtocol(phantom);
        System.out.println("Протокол «Сброс фантомного нагрева» добавлен.");
    }
